    29→/* =========================================================================
    30→   Contact modal — form + contact info
    31→   ========================================================================= */
    32→export function ContactModal({
    33→  children,
    34→}: {
    35→  children?: React.ReactNode;
    36→}) {
    37→  const { toast } = useToast();
    38→  const [open, setOpen] = useState(false);
    39→  const [loading, setLoading] = useState(false);
    40→  const [form, setForm] = useState({
    41→    name: "",
    42→    email: "",
    43→    phone: "",
    44→    subject: "",
    45→    message: "",
    46→  });
    47→
    48→  const handleSubmit = async (e: React.FormEvent) => {
    49→    e.preventDefault();
    50→    setLoading(true);
    51→    try {
    52→      const res = await fetch("/api/contact", {
    53→        method: "POST",
    54→        headers: { "Content-Type": "application/json" },
    55→        body: JSON.stringify(form),
    56→      });
    57→      const data = await res.json();
    58→      if (!res.ok || !data.ok) {
    59→        throw new Error(data.error || "Erreur d'envoi");
    60→      }
    61→      toast({
    62→        title: "Message envoyé !",
    63→        description: data.message,
    64→      });
    65→      setForm({ name: "", email: "", phone: "", subject: "", message: "" });
    66→      setOpen(false);
    67→    } catch (err) {
    68→      toast({
    69→        title: "Erreur d'envoi",
    70→        description:
    71→          err instanceof Error
    72→            ? err.message
    73→            : "Veuillez réessayer ou nous écrire à contact@gatopouch.com",
    74→        variant: "destructive",
    75→      });
    76→    } finally {
    77→      setLoading(false);
    78→    }
    79→  };
    80→
    81→  return (
    82→    <Dialog open={open} onOpenChange={setOpen}>
    83→      <DialogTrigger asChild>
    84→        {children || (
    85→          <button className="hover:text-peach-300 transition-colors">
    86→            Contact
    87→          </button>
    88→        )}
    89→      </DialogTrigger>
    90→      <DialogContent className="max-w-3xl max-h-[90vh] overflow-y-auto bg-cream-50 border-peach-400/30">
    91→        <DialogHeader>
    92→          <DialogTitle className="font-display text-2xl md:text-3xl text-cinnamon-900">
    93→            Contactez GatoPouch
    94→          </DialogTitle>
    95→          <DialogDescription className="text-cinnamon-700">
    96→            Une question sur le GatoPouch, votre commande, ou une demande
    97→            partenariat ? Écrivez-nous, nous répondons sous 24 à 48h ouvrées.
    98→          </DialogDescription>
    99→        </DialogHeader>
   100→

   102→          {/* Contact form (3/5) */}
   103→          <form onSubmit={handleSubmit} className="md:col-span-3 space-y-3">
   104→            <div className="grid sm:grid-cols-2 gap-3">
   105→              <div className="space-y-1.5">

   107→                  Nom *
   108→                </Label>

   110→                  id="name"
   111→                  required
   112→                  value={form.name}
   113→                  onChange={(e) => setForm({ ...form, name: e.target.value })}
   114→                  className="bg-white border-cinnamon-900/15 focus-visible:ring-peach-400"
   115→                  placeholder="Camille Dubois"
